/* KME Apps tracker. Add to any app:
   <script src="https://YOUR-KME-WORKER/kme.js" data-app="your-app-name" defer></script>
   Counts one visit per page load, reports up to 5 errors per page, and keeps a random device id
   (no names, emails or IP addresses). Optional: KME.event('Signed up') for your own milestones. */
(function () {
  var s = document.currentScript, app = s && s.getAttribute('data-app');
  if (!app || window.KME) return;
  var END = s.src.replace(/\/kme\.js.*$/, '') + '/t', dev = '';
  try { dev = localStorage.getItem('kme_d') || ''; if (!dev) { dev = Math.random().toString(36).slice(2) + Date.now().toString(36); localStorage.setItem('kme_d', dev); } } catch (e) {}
  var ua = navigator.userAgent || '';
  var plat = /iPhone|iPad|iPod/.test(ua) ? 'ios' : /Android/.test(ua) ? 'android' : 'web';
  try { if (window.Capacitor && window.Capacitor.isNativePlatform && window.Capacitor.isNativePlatform()) plat += '-app'; else if (window.matchMedia && matchMedia('(display-mode: standalone)').matches || navigator.standalone) plat += '-home'; } catch (e) {}
  function send(o) {
    o.app = app; o.dev = dev; o.plat = plat;
    o.path = location.pathname;
    try { var r = document.referrer ? new URL(document.referrer).host : ''; o.ref = r && r !== location.host ? r : ''; } catch (e) {}
    var body = JSON.stringify(o);
    try { if (navigator.sendBeacon && navigator.sendBeacon(END, new Blob([body], { type: 'text/plain' }))) return; } catch (e) {}
    try { fetch(END, { method: 'POST', body: body, keepalive: true, mode: 'no-cors', headers: { 'content-type': 'text/plain' } }); } catch (e) {}
  }
  send({ kind: 'view' });
  var n = 0, seen = {};
  function err(m, src) {
    m = String(m || '').slice(0, 300);
    if (!m || n >= 5 || seen[m]) return;
    seen[m] = 1; n++;
    send({ kind: 'error', msg: m, src: String(src || '').slice(0, 200) });
  }
  addEventListener('error', function (e) { if (e && e.message) err(e.message, (String(e.filename || '').split('/').pop() || '') + (e.lineno ? ':' + e.lineno : '')); });
  addEventListener('unhandledrejection', function (e) { var r = e && e.reason; err((r && (r.message || (typeof r === 'string' ? r : ''))) || 'Unhandled promise rejection', r && r.stack ? String(r.stack).split('\n')[1] : ''); });
  window.KME = { event: function (name) { send({ kind: 'event', msg: String(name || '').slice(0, 80) }); } };
})();
